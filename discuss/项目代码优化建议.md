```python
class ImagePreprocessor:
    """图像预处理器"""

    @staticmethod
    def blur(frame: np.ndarray, kernel_size: int = 5) -> np.ndarray:
        """高斯模糊预处理"""
        return cv2.GaussianBlur(frame, (kernel_size, kernel_size), 0)

    @staticmethod
    def denoise(frame: np.ndarray, strength: int = 10) -> np.ndarray:
        """降噪预处理"""
        return cv2.fastNlMeansDenoisingColored(frame, None, strength, strength, 7, 21)

    @staticmethod
    def sharpen(frame: np.ndarray) -> np.ndarray:
        """锐化预处理"""
        kernel = np.array([[-1,-1,-1], [-1,9,-1], [-1,-1,-1]])
        return cv2.filter2D(frame, -1, kernel)


class ImagePostprocessor:
    """图像后处理器"""

    @staticmethod
    def smooth_edges(frame: np.ndarray, mask: np.ndarray,
                     blur_radius: int = 5) -> np.ndarray:
        """修复区域边缘平滑"""
        # 对修复区域边缘进行羽化处理
        dilated_mask = cv2.dilate(mask, np.ones((5,5), np.uint8), iterations=2)
        edge_mask = dilated_mask - mask
        blurred = cv2.GaussianBlur(frame, (blur_radius*2+1, blur_radius*2+1), 0)

        # 边缘区域混合
        alpha = edge_mask.astype(float) / 255.0
        result = frame * (1 - alpha[:,:,np.newaxis]) + blurred * alpha[:,:,np.newaxis]
        return result.astype(np.uint8)

    @staticmethod
    def enhance_contrast(frame: np.ndarray, alpha: float = 1.2) -> np.ndarray:
        """增强对比度"""
        return cv2.convertScaleAbs(frame, alpha=alpha, beta=0)
```

### 3.2 🟡 中优先级问题

#### 问题3: GPU内存无监控

**位置**: `app/core/ai/yolo_detector.py`, `dl_inpainter.py`

**现状**: 批处理时未检查GPU显存，高分辨率视频可能导致CUDA OOM

**优化建议**:
```python
import torch

class GPUMemoryMonitor:
    """GPU显存监控器"""

    MIN_FREE_MB = 500  # 最小保留显存

    @classmethod
    def check_available(cls) -> float:
        """获取可用显存(MB)"""
        if torch.cuda.is_available():
            free, total = torch.cuda.mem_get_info()
            return free / (1024 ** 2)
        return float('inf')

    @classmethod
    def ensure_memory(cls, required_mb: float = 500) -> bool:
        """确保有足够显存，必要时清理缓存"""
        available = cls.check_available()
        if available < required_mb:
            torch.cuda.empty_cache()
            available = cls.check_available()
        return available >= required_mb

    @classmethod
    def adaptive_batch_size(cls, base_size: int, frame_size: tuple) -> int:
        """根据显存自适应调整批大小"""
        available_mb = cls.check_available()
        # 估算每帧显存占用 (假设 1080p ≈ 50MB)
        est_per_frame = (frame_size[0] * frame_size[1] * 3) / (1024 ** 2) * 10
        safe_batch = int(available_mb * 0.7 / est_per_frame)
        return max(1, min(base_size, safe_batch))
```

**使用示例**:
```python
# 在 yolo_detector.py 中
def detect_batch(self, frames: List[np.ndarray]) -> List[np.ndarray]:
    # 自适应批大小
    batch_size = GPUMemoryMonitor.adaptive_batch_size(
        base_size=self.batch_size,
        frame_size=frames[0].shape[:2]
    )

    results = []
    for i in range(0, len(frames), batch_size):
        batch = frames[i:i+batch_size]
        # 确保有足够显存
        if not GPUMemoryMonitor.ensure_memory(500):
            raise RuntimeError("GPU显存不足，请减小视频分辨率或批大小")
        results.extend(self._detect(batch))
    return results
```

#### 问题4: 流水线有序性开销大

**位置**: `app/core/video/pipeline_processor.py`

**现状**:
```python

while next_frame_index in frame_buffer:
    out.write(frame_buffer.pop(next_frame_index))
    next_frame_index += 1
```

**问题**: 字典查找 + 频繁插入删除，且需要等待缺失帧

**优化建议**: 使用优先级队列
```python
from heapq import heappush, heappop
from dataclasses import dataclass, field
from typing import Any

@dataclass(order=True)
class PrioritizedFrame:
    index: int
    frame: Any = field(compare=False)

class OrderedFrameWriter:
    """有序帧写入器"""

    def __init__(self, writer: cv2.VideoWriter):
        self.writer = writer
        self.heap = []
        self.next_index = 0

    def add_frame(self, index: int, frame: np.ndarray):
        """添加帧到缓冲"""
        heappush(self.heap, PrioritizedFrame(index, frame))
        self._flush_ready()

    def _flush_ready(self):
        """写入所有已就绪的帧"""
        while self.heap and self.heap[0].index == self.next_index:
            item = heappop(self.heap)
            self.writer.write(item.frame)
            self.next_index += 1

    def flush_all(self):
        """强制写入所有缓冲帧（处理结束时）"""
        while self.heap:
            item = heappop(self.heap)
            self.writer.write(item.frame)
```

#### 问题5: 背压控制粗糙

**位置**: `app/core/video/pipeline_processor.py`

**现状**:
```python
while len(frame_buffer) >= MAX_BUFFER_SIZE:
    time.sleep(0.01)  # 固定等待10ms
```

**优化建议**: 使用条件变量实现精确背压
```python
import threading

class BackpressureController:
    """背压控制器"""

    def __init__(self, high_watermark: int, low_watermark: int):
        self.high_watermark = high_watermark
        self.low_watermark = low_watermark
        self.current_size = 0
        self.lock = threading.Lock()
        self.not_full = threading.Condition(self.lock)
        self.not_empty = threading.Condition(self.lock)

    def acquire_slot(self):
        """生产者：获取槽位（可能阻塞）"""
        with self.not_full:
            while self.current_size >= self.high_watermark:
                self.not_full.wait()
            self.current_size += 1

    def release_slot(self):
        """消费者：释放槽位"""
        with self.not_empty:
            self.current_size -= 1
            if self.current_size <= self.low_watermark:
                self.not_full.notify_all()
```

```python
from dataclasses import dataclass
from typing import Any, Optional

@dataclass
class ConfigValidator:
    """配置参数验证器"""

    RULES = {
        'conf_threshold': {'type': float, 'min': 0.0, 'max': 1.0},
        'iou_threshold': {'type': float, 'min': 0.0, 'max': 1.0},
        'batch_size': {'type': int, 'min': 1, 'max': 64},
        'num_workers': {'type': int, 'min': 1, 'max': 16},
    }

    @classmethod
    def validate(cls, key: str, value: Any) -> tuple[bool, Optional[str]]:
        """验证配置值"""
        if key not in cls.RULES:
            return True, None

        rule = cls.RULES[key]

        # 类型检查
        if not isinstance(value, rule['type']):
            return False, f"{key} 必须是 {rule['type'].__name__} 类型"

        # 范围检查
        if 'min' in rule and value < rule['min']:
            return False, f"{key} 不能小于 {rule['min']}"
        if 'max' in rule and value > rule['max']:
            return False, f"{key} 不能大于 {rule['max']}"

        return True, None
```

```python
import requests
from tenacity import retry, stop_after_attempt, wait_exponential

class RobustModelDownloader:
    """健壮的模型下载器"""

    TIMEOUT = 30  # 连接超时
    CHUNK_SIZE = 8192

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=4, max=60)
    )
    def download(self, url: str, dest_path: str,
                 progress_callback=None) -> bool:
        """下载模型文件，支持断点续传"""

        # 检查已下载大小
        downloaded_size = 0
        if os.path.exists(dest_path):
            downloaded_size = os.path.getsize(dest_path)

        headers = {'Range': f'bytes={downloaded_size}-'} if downloaded_size else {}

        response = requests.get(
            url,
            headers=headers,
            stream=True,
            timeout=self.TIMEOUT
        )

        total_size = int(response.headers.get('content-length', 0)) + downloaded_size
        mode = 'ab' if downloaded_size else 'wb'

        with open(dest_path, mode) as f:
            for chunk in response.iter_content(chunk_size=self.CHUNK_SIZE):
                if chunk:
                    f.write(chunk)
                    downloaded_size += len(chunk)
                    if progress_callback:
                        progress_callback(downloaded_size, total_size)

        return True
```

#### 精确背压控制
```python
# backpressure.py - Condition变量实现精确等待
class BackpressureController:
    def __init__(self, high_watermark: int = 50, low_watermark: int = 25):
        self._lock = threading.Lock()
        self._not_full = threading.Condition(self._lock)

    def acquire_slot(self) -> bool:
        with self._not_full:
            while self._current_size >= self.high_watermark:
                self._not_full.wait(timeout=self.timeout)  # 精确阻塞
            self._current_size += 1

    def release_slot(self) -> None:
        with self._not_full:
            self._current_size -= 1
            if self._current_size <= self.low_watermark:
                self._not_full.notify_all()  # 唤醒等待者
```
